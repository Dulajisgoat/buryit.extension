// popup.js — toolbar popup controller. Queries the active tab, renders
// per-site state, and relays commands to the content script via background.js.

(function () {
  "use strict";

  var hostnameEl = document.getElementById("hostname");
  var toggleEl = document.getElementById("enabledToggle");
  var countEl = document.getElementById("count");
  var resetEl = document.getElementById("resetButton");
  var statusEl = document.getElementById("status");

  var activeTabId = null;

  function setStatus(msg, isError) {
    statusEl.textContent = msg || "";
    statusEl.classList.toggle("buryit-status-error", !!isError);
  }

  function showDisabledState(msg) {
    hostnameEl.textContent = "—";
    toggleEl.checked = false;
    toggleEl.disabled = true;
    countEl.textContent = "0 element(s) buried";
    resetEl.disabled = true;
    setStatus(msg || "BuryIt isn't active on this page.", true);
  }

  function formatCount(stored, matching) {
    if (typeof matching === "number" && matching !== stored) {
      return stored + " buried · " + matching + " hidden now";
    }
    return stored + (stored === 1 ? " element buried" : " element(s) buried");
  }

  function sendRelay(payload, cb) {
    if (activeTabId === null) { cb({ ok: false }); return; }
    try {
      chrome.runtime.sendMessage({ type: "relay", tabId: activeTabId, payload: payload }, function (r) {
        if (chrome.runtime.lastError) { cb({ ok: false, error: chrome.runtime.lastError.message }); return; }
        cb(r || { ok: false });
      });
    } catch (e) { cb({ ok: false, error: String(e) }); }
  }

  function refresh() {
    sendRelay({ type: "get-state" }, function (state) {
      if (!state || !state.ok) {
        showDisabledState("Couldn't reach this page (try reloading the tab).");
        return;
      }
      sendRelay({ type: "count-matching" }, function (m) {
        var matching = m && m.ok ? m.count : undefined;
        hostnameEl.textContent = state.hostname;
        toggleEl.checked = !!state.enabled;
        toggleEl.disabled = false;
        countEl.textContent = formatCount(state.count, matching);
        resetEl.disabled = state.count === 0;
        setStatus("", false);
      });
    });
  }

  function init() {
    try {
      chrome.tabs.query({ active: true, currentWindow: true }, function (tabs) {
        if (chrome.runtime.lastError || !tabs || !tabs.length || typeof tabs[0].id !== "number") {
          showDisabledState("No active tab found.");
          return;
        }
        var url = tabs[0].url || "";
        if (url.indexOf("chrome://") === 0 || url.indexOf("chrome-extension://") === 0 || url.indexOf("edge://") === 0) {
          showDisabledState("BuryIt doesn't run on browser pages.");
          return;
        }
        activeTabId = tabs[0].id;
        refresh();
      });
    } catch (e) {
      showDisabledState("Something went wrong opening this popup.");
    }
  }

  toggleEl.addEventListener("change", function () {
    var value = toggleEl.checked;
    toggleEl.disabled = true;
    sendRelay({ type: "set-enabled", value: value }, function (r) {
      if (!r || !r.ok) { showDisabledState("Couldn't update this site."); return; }
      refresh();
    });
  });

  resetEl.addEventListener("click", function () {
    resetEl.disabled = true;
    sendRelay({ type: "reset" }, function (r) {
      if (!r || !r.ok) { showDisabledState("Couldn't reset this site."); return; }
      refresh();
    });
  });

  document.addEventListener("DOMContentLoaded", init);
})();
